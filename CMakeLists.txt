cmake_minimum_required(VERSION 3.22)
project(PazurAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# Debug APKs keep symbols/assertions, but their physics/audio dependencies also
# need optimization on low-end phones. Deliberately avoid -ffast-math/NDEBUG.
option(PAZUR_OPTIMIZE_DEPENDENCIES "Optimize native dependencies in debug builds" ON)
if(PAZUR_OPTIMIZE_DEPENDENCIES)
  add_compile_options(-O2)
endif()
# Cross-file optimization retains assertions and normal floating-point semantics.
# Reversible with -PnativeLto=OFF for measured comparisons.
option(PAZUR_NATIVE_IPO "Use ThinLTO and O3 for native code" ON)
if(PAZUR_NATIVE_IPO)
  add_compile_options(-O3 -flto=thin)
  add_link_options(-flto=thin)
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(upstream/SDL)
# Static add-ons can share the single SDL runtime owned by SDLActivity.
add_library(SDL2::SDL2-static INTERFACE IMPORTED)
set_target_properties(SDL2::SDL2-static PROPERTIES INTERFACE_LINK_LIBRARIES SDL2::SDL2)
set(SDL2IMAGE_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL2TTF_INSTALL OFF CACHE BOOL "" FORCE)
foreach(format AVIF JXL TIF WEBP)
  set(SDL2IMAGE_${format} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(upstream/SDL_image)
foreach(format FLAC GME MOD MP3 OPUS WAVPACK)
  set(SDL2MIXER_${format} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL2MIXER_MIDI ON CACHE BOOL "" FORCE)
set(SDL2MIXER_MIDI_TIMIDITY ON CACHE BOOL "" FORCE)
set(SDL2MIXER_MIDI_FLUIDSYNTH OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_VORBIS "" CACHE STRING "" FORCE)
set(SDL2MIXER_SAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(upstream/SDL_mixer)
set(SDL2TTF_VENDORED ON CACHE BOOL "" FORCE)
set(SDL2TTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
set(SDL2TTF_SAMPLES OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
add_subdirectory(upstream/SDL_ttf)
set(OC ${CMAKE_CURRENT_SOURCE_DIR}/upstream/OpenClaw)
add_subdirectory(${OC}/Box2D box2d)
add_subdirectory(${OC}/libwap libwap)
add_subdirectory(${OC}/ThirdParty/Tinyxml tinyxml)
add_library(SDL2_gfx STATIC upstream/SDL2_gfx/SDL2_gfxPrimitives.c upstream/SDL2_gfx/SDL2_rotozoom.c)
target_link_libraries(SDL2_gfx SDL2)

# Upstream uses SDL2/header.h, while SDL sources expose flat include directories.
set(COMPAT ${CMAKE_CURRENT_BINARY_DIR}/compat)
file(MAKE_DIRECTORY ${COMPAT}/SDL2)
file(GLOB SDL_HEADERS upstream/SDL/include/*.h upstream/SDL_image/include/*.h upstream/SDL_image/*.h upstream/SDL_mixer/include/*.h upstream/SDL_ttf/*.h upstream/SDL2_gfx/*.h)
file(COPY ${SDL_HEADERS} DESTINATION ${COMPAT}/SDL2)
file(GLOB_RECURSE GAME_SOURCES CONFIGURE_DEPENDS ${OC}/OpenClaw/*.cpp)
list(FILTER GAME_SOURCES EXCLUDE REGEX "/main.cpp$")
add_library(openclaw SHARED ${GAME_SOURCES} native/android_main.cpp native/TouchControls.cpp native/LavaGeyserComponent.cpp native/OmarEncounterComponent.cpp native/CombatProbe.cpp native/PlatformInput.cpp native/GamepadControls.cpp)
target_sources(openclaw PRIVATE native/HapticFeedback.cpp native/ComponentProfiler.cpp)
target_include_directories(openclaw PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/native ${COMPAT} ${OC}/OpenClaw ${OC}/Box2D ${OC}/libwap ${OC}/ThirdParty ${OC}/ThirdParty/Tinyxml)
target_compile_definitions(openclaw PRIVATE ANDROID SDL_MAIN_HANDLED)
target_compile_options(openclaw PRIVATE -O2 -Wno-deprecated-declarations)
if(PAZUR_NATIVE_IPO)
  target_compile_options(openclaw PRIVATE -O3 -flto=thin)
endif()
target_link_libraries(openclaw PRIVATE libwap tinyxml Box2D SDL2 SDL2_image SDL2_mixer SDL2_ttf SDL2_gfx log android GLESv2)
